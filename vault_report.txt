#!/bin/bash

# Define paths
VAULT_DIR="$HOME/secure_vault"
REPORT_FILE="$VAULT_DIR/vault_report.txt"

# 1. Print header to the terminal
echo "Generating Vault Report..."

# 2. Initialize the report file with headers (using > to overwrite)
echo "--- SECURITY MONITORING REPORT ---" > "$REPORT_FILE"
echo "Date: $(date)" >> "$REPORT_FILE"
echo "--------------------------------" >> "$REPORT_FILE"

# 3. Loop through each file in the vault
for file in "$VAULT_DIR"/*; do
    # Skip the report file itself
    if [[ "$file" == "$REPORT_FILE" ]]; then
        continue
    fi

    filename=$(basename "$file")
    # Get numeric permissions (e.g., 600, 644)
    permissions=$(stat -c "%a" "$file")

    # Write file details to the report
    {
        echo "File: $filename"
        echo "Permissions: $permissions"
        
        # Check if permissions are more open than 644
        if [ "$permissions" -gt 644 ]; then
            echo "⚠️ SECURITY RISK DETECTED"
        else
            echo "Status: Secure"
        fi
        echo "--------------------------------"
    } >> "$REPORT_FILE"
done

# 4. Print confirmation that the report has been created
echo "Confirmation: The report has been created at $REPORT_FILE"
